cmake_minimum_required(VERSION 3.8)
project(robot_ros)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp robot_msgs sensor_msgs nav_msgs geometry_msgs rosgraph_msgs std_msgs std_srvs ament_index_cpp robot_interface state_estimator navigation controller learning_policy)
  find_package(${dep} REQUIRED)
endforeach()
# RoboStack exports Python generators through C++ targets on macOS.
if(APPLE)
  get_property(ros_targets DIRECTORY PROPERTY IMPORTED_TARGETS)
  foreach(ros_target IN LISTS ros_targets)
    get_target_property(links ${ros_target} INTERFACE_LINK_LIBRARIES)
    if(links)
      list(FILTER links EXCLUDE REGEX "__rosidl_generator_py$")
      set_property(TARGET ${ros_target} PROPERTY INTERFACE_LINK_LIBRARIES "${links}")
    endif()
  endforeach()
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(GLFW3 REQUIRED IMPORTED_TARGET glfw3)
set(MUJOCO_FRAMEWORK_DIR "/Applications/MuJoCo.app/Contents/Frameworks" CACHE PATH "MuJoCo framework directory")
find_file(MUJOCO_DYLIB NAMES libmujoco.3.12.0.dylib PATHS "${MUJOCO_FRAMEWORK_DIR}/mujoco.framework/Versions/A" REQUIRED)
add_library(ros_backend STATIC ../mujoco_backend/src/mujoco_backend.cpp
  ../mujoco_backend/src/mujoco_viewer.cpp ../mujoco_backend/src/known_map_world.cpp)
target_include_directories(ros_backend PUBLIC ../mujoco_backend/include)
target_compile_options(ros_backend PUBLIC "-F${MUJOCO_FRAMEWORK_DIR}")
target_link_libraries(ros_backend "${MUJOCO_DYLIB}" PkgConfig::GLFW3)
ament_target_dependencies(ros_backend robot_interface navigation)
include_directories(include)
foreach(node backend_node estimator_node mpc_node ppo_node direct_reference)
  add_executable(${node} src/${node}.cpp)
  target_link_libraries(${node} rclcpp::rclcpp ament_index_cpp::ament_index_cpp)
  foreach(msg robot_msgs sensor_msgs nav_msgs geometry_msgs rosgraph_msgs std_msgs std_srvs)
    target_link_libraries(${node} ${msg}::${msg}__rosidl_typesupport_cpp)
  endforeach()
  ament_target_dependencies(${node} robot_interface navigation)
  set_target_properties(${node} PROPERTIES BUILD_RPATH "${MUJOCO_FRAMEWORK_DIR}" INSTALL_RPATH "${MUJOCO_FRAMEWORK_DIR};${GLFW3_LIBRARY_DIRS};@loader_path/../../../robot_msgs/lib")
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
target_link_libraries(backend_node ros_backend)
target_link_libraries(direct_reference ros_backend)
ament_target_dependencies(estimator_node state_estimator)
ament_target_dependencies(mpc_node controller)
ament_target_dependencies(ppo_node learning_policy)
ament_target_dependencies(direct_reference controller state_estimator)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES ../../../model/Robot1.xml ../../../model/known_map.txt DESTINATION share/${PROJECT_NAME}/model)
install(FILES ../../../artifacts/learning/selected_policy.txt DESTINATION share/${PROJECT_NAME}/policy OPTIONAL)
ament_package()
